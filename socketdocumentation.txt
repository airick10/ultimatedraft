Install Socket
	pip install flask-socketio eventlet

-------------------------
Add to requirements.txt
	flask-socketio==5.3.6
	eventlet==0.35.2

---------------------------
Set this up in __init__.py
from flask import Flask
from flask_socketio import SocketIO

socketio = SocketIO()

def create_app():
    app = Flask(__name__)

    from .routes import main
    app.register_blueprint(main)

    socketio.init_app(app)

    return app
----------------------------
Update wsgi.py
from ultimatebaseballdraft import create_app, socketio

app = create_app()

if __name__ == "__main__":
    socketio.run(app, debug=True, port=5050)
"This is important — you run socketio.run() instead of app.run() so the WebSocket server starts correctly."

----------------------------
Add SocketIO JS client to base.html:
<script src="https://cdn.socket.io/4.7.2/socket.io.min.js"></script>

----------------------------
